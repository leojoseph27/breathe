   130→   130→def extract_features(data):
   131→   131→    result = np.array([])
   132→   132→
   133→   133→    # Ensure data has sufficient length
   134→   134→    if len(data) < 100:
   135→   135→        # Return a default feature vector for very short audio
   136→   136→        return np.zeros(162)  # Default size to match expected input
   137→   137→
   138→   138→    # Zero Crossing Rate
   139→   139→    try:
   140→   140→        zcr = np.mean(librosa.feature.zero_crossing_rate(data).T, axis=0)
   141→   141→        result = np.hstack((result, zcr))
   142→   142→    except:
   143→   143→        result = np.hstack((result, np.zeros(1)))  # Default value if ZCR fails
   144→   144→
   145→   145→    # Short-time Fourier Transform and Chroma features
   146→   146→    try:
   147→   147→        stft = np.abs(librosa.stft(data))
   148→   148→        # Suppress the pitch tuning warning that causes the error
   149→   149→        import warnings
   150→   150→        with warnings.catch_warnings():
   151→   151→            warnings.simplefilter("ignore")
   152→   152→            chroma = np.mean(librosa.feature.chroma_stft(S=stft, sr=22050, n_chroma=12).T, axis=0)
   153→   153→        result = np.hstack((result, chroma))
   154→   154→    except:
   155→   155→        result = np.hstack((result, np.zeros(12)))  # Default 12 chroma features if it fails
   156→   156→
   157→   157→    # MFCC features
   158→   158→    try:
   159→   159→        # Suppress the pitch tuning warning that causes the error
   160→   160→        import warnings
   161→   161→        with warnings.catch_warnings():
   162→   162→            warnings.simplefilter("ignore")
   163→   163→            mfcc = np.mean(librosa.feature.mfcc(y=data, sr=22050, n_mfcc=20).T, axis=0)
   164→   164→        result = np.hstack((result, mfcc))
   165→   165→    except:
   166→   166→        # If MFCC fails, add zeros as fallback
   167→   167→        result = np.hstack((result, np.zeros(20)))  # Assuming 20 MFCC coefficients
   168→   168→
   169→   169→    # RMS
   170→   170→    try:
   171→   171→        rms = np.mean(librosa.feature.rms(y=data).T, axis=0)
   172→   172→        result = np.hstack((result, rms))
   173→   173→    except:
   174→   174→        result = np.hstack((result, np.zeros(1)))  # Default value if RMS fails
   175→   175→
   176→   176→    # Mel spectrogram
   177→   177→    try:
   178→   178→        # Suppress the pitch tuning warning that causes the error
   179→   179→        import warnings
   180→   180→        with warnings.catch_warnings():
   181→   181→            warnings.simplefilter("ignore")
   182→   182→            mel = np.mean(librosa.feature.melspectrogram(y=data, sr=22050, n_mels=128).T, axis=0)
   183→   183→        result = np.hstack((result, mel))
   184→   184→    except:
   185→   185→        result = np.hstack((result, np.zeros(128)))  # Default 128 mel bands if it fails
   186→   186→
   187→   187→    return result

   189→   189→def preprocess_audio(audio_path):
   190→   190→    try:
   191→   191→        data, sr = librosa.load(audio_path, duration=2.5, offset=0.6, mono=True)
   192→   192→        # Ensure we have enough samples to process
   193→   193→        if len(data) < 100:
   194→   194→            raise ValueError("Audio file too short for processing")
   195→   195→        features = extract_features(data)
   196→   196→        features = features.reshape(1, -1)
   197→   197→        features = np.expand_dims(features, axis=2)
   198→   198→        return features
   199→   199→    except Exception as e:
   200→   200→        print(f"Error preprocessing audio: {e}")
   201→   201→        # Return a dummy feature vector to avoid crashes
   202→   202→        dummy_features = np.zeros((1, 162, 1))  # Shape matches model expectation
   203→   203→        return dummy_features
