   126→def allowed_file(filename):
   127→    return '.' in filename and \
   128→           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

   130→def extract_features(data):
   131→    result = np.array([])
   132→
   133→    # Ensure data has sufficient length
   134→    if len(data) < 100:
   135→        # Return a default feature vector for very short audio
   136→        return np.zeros(162)  # Default size to match expected input
   137→
   138→    # Zero Crossing Rate
   139→    try:
   140→        zcr = np.mean(librosa.feature.zero_crossing_rate(data).T, axis=0)
   141→        result = np.hstack((result, zcr))
   142→    except:
   143→        result = np.hstack((result, np.zeros(1)))  # Default value if ZCR fails
   144→
   145→    # Short-time Fourier Transform and Chroma features
   146→    try:
   147→        stft = np.abs(librosa.stft(data))
   148→        # Suppress the pitch tuning warning that causes the error
   149→        import warnings
   150→        with warnings.catch_warnings():
   151→            warnings.simplefilter("ignore")
   152→            chroma = np.mean(librosa.feature.chroma_stft(S=stft, sr=22050, n_chroma=12).T, axis=0)
   153→        result = np.hstack((result, chroma))
   154→    except:
   155→        result = np.hstack((result, np.zeros(12)))  # Default 12 chroma features if it fails
   156→
   157→    # MFCC features
   158→    try:
   159→        # Suppress the pitch tuning warning that causes the error
   160→        import warnings
   161→        with warnings.catch_warnings():
   162→            warnings.simplefilter("ignore")
   163→            mfcc = np.mean(librosa.feature.mfcc(y=data, sr=22050, n_mfcc=20).T, axis=0)
   164→        result = np.hstack((result, mfcc))
   165→    except:
   166→        # If MFCC fails, add zeros as fallback
   167→        result = np.hstack((result, np.zeros(20)))  # Assuming 20 MFCC coefficients
   168→
   169→    # RMS
   170→    try:
   171→        rms = np.mean(librosa.feature.rms(y=data).T, axis=0)
   172→        result = np.hstack((result, rms))
   173→    except:
   174→        result = np.hstack((result, np.zeros(1)))  # Default value if RMS fails
   175→
   176→    # Mel spectrogram
   177→    try:
   178→        # Suppress the pitch tuning warning that causes the error
   179→        import warnings
   180→        with warnings.catch_warnings():
   181→            warnings.simplefilter("ignore")
   182→            mel = np.mean(librosa.feature.melspectrogram(y=data, sr=22050, n_mels=128).T, axis=0)
   183→        result = np.hstack((result, mel))
   184→    except:
   185→        result = np.hstack((result, np.zeros(128)))  # Default 128 mel bands if it fails
   186→
   187→    return result

   189→def preprocess_audio(audio_path):
   190→    try:
   191→        data, sr = librosa.load(audio_path, duration=2.5, offset=0.6, mono=True)
   192→        # Ensure we have enough samples to process
   193→        if len(data) < 100:
   194→            raise ValueError("Audio file too short for processing")
   195→        features = extract_features(data)
   196→        features = features.reshape(1, -1)
   197→        features = np.expand_dims(features, axis=2)
   198→        return features
   199→    except Exception as e:
   200→        print(f"Error preprocessing audio: {e}")
   201→        # Return a dummy feature vector to avoid crashes
   202→        dummy_features = np.zeros((1, 162, 1))  # Shape matches model expectation
   203→        return dummy_features

   242→def predict_asthma(features_dict):
   243→    if not ASTHMA_MODEL_LOADED:
   244→        print("Error: Asthma model not loaded")
   245→        return None, 0
   246→    
   247→    # Convert features to array in the same order as training
   248→    feature_order = [
   249→        'age', 'gender', 'bmi', 'smoking', 'familyHistory', 
   250→        'allergyHistory', 'lungFunctionFeV1', 'wheezing', 
   251→        'shortnessOfBreath', 'chestTightness'
   252→    ]
   253→    
   254→    try:
   255→        # Create feature array
   256→        features = np.array([[features_dict[key] for key in feature_order]])
   257→        
   258→        # Get prediction and probability
   259→        prediction = asthma_model.predict(features)[0]
   260→        probabilities = asthma_model.predict_proba(features)[0]
   261→        confidence = max(probabilities) * 100
   262→        
   263→        return int(prediction), round(confidence, 2)
   264→    except Exception as e:
   265→        print(f"Error in asthma prediction: {e}")
   266→        return None, 0

   324→        return jsonify({'error': 'No file selected'}), 400
   325→

   332→            filepath = os.path.join(temp_dir, temp_filename)
   333→            
   334→            file.save(filepath)
   335→            
   336→            # Make prediction
   337→            if not AUDIO_MODEL_LOADED:
   338→                print("Warning: Audio model not loaded due to version incompatibility, cannot make accurate predictions")
   339→                
   340→            pred, conf = predict_disease(filepath)
   341→            
   342→            # Check if it's an error condition
   343→            if pred == "Model Error - Version Incompatible":
   344→                return jsonify({
   345→                    'error': 'Model could not be loaded due to version incompatibility. The model file was saved with a different version of Keras/TensorFlow. Please use a compatible version or re-save the model.',
   346→                    'prediction': None,
   347→                    'confidence': None
   348→                }), 500
   349→            
   350→            # Clean up temporary file
   351→            os.remove(filepath)
   352→            
   353→            return jsonify({
   354→                'prediction': pred
   355→            })
   356→        except Exception as e:
   357→            return jsonify({'error': f'Error processing audio: {str(e)}'}), 500
